pip install django-sitemaps

# sitemaps.py
from django.contrib.sitemaps import Sitemap
from .models import Course, TutorProfile, CityPage

class CourseSitemap(Sitemap):
    changefreq = "weekly"
    priority = 0.9
    def items(self):
        return Course.objects.filter(is_active=True)

class CityTutorSitemap(Sitemap):
    changefreq = "daily"
    priority = 0.95  # highest priority
    def items(self):
        return CityPage.objects.all()

# urls.py
sitemaps = {
  'courses': CourseSitemap,
  'city': CityTutorSitemap,
}
path('sitemap.xml', sitemap, {'sitemaps': sitemaps})

AI-Powered Content at Scale with Django
Strategy: Use Claude/OpenAI API to generate city-specific content
Create unique, SEO-rich content for every city × subject × exam combination automatically. 500 cities × 10 subjects = 5,000 unique pages in days.
Django Management Command — Bulk AI Content Generator
python manage.py generate_city_content --city=Ahmedabad

# management/commands/generate_city_content.py
import anthropic
client = anthropic.Anthropic(api_key=settings.CLAUDE_API_KEY)

def generate_tutor_content(city, subject):
    prompt = f"""Write SEO content for home tutors in {city} for {subject}.
    Include: local relevance, parent pain points, Ranjee benefits.
    Format: H2 intro, 3 paragraphs, FAQ section (5 Q&A).
    Target keyword: 'home tutor in {city} for {subject}'"""
    
    msg = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1500,
        messages=[{"role":"user","content":prompt}]
    )
    CityPage.objects.update_or_create(
        city=city, subject=subject,
        defaults={'content': msg.content[0].text}
    )

Student Referral System as an SEO Multiplier
Referral = Free Backlinks + User Generated Content
Every referred student creates a shareable page that Google indexes
1
Shareable referral landing page: ranjee.com/ref/priya-sharma — Each student gets a unique page with their name, city, course. Google indexes thousands of these.
2
Auto-generated review pages: After completing a course, trigger AI to create a testimonial page — "Priya from Ahmedabad completed Python course"
3
WhatsApp share button on every referral page — students share to groups. Each share = traffic + social signal to Google.
4
Referral leaderboard page: ranjee.com/top-students/ahmedabad — ranks top referrers. Students share it. More organic backlinks.
Django Referral Model
class ReferralPage(models.Model):
    student = models.ForeignKey(User, on_delete=models.CASCADE)
    ref_code = models.CharField(max_length=20, unique=True)
    city = models.CharField(max_length=50)
    course = models.ForeignKey(Course, on_delete=models.SET_NULL, null=True)
    # SEO fields (AI-generated)
    meta_title = models.CharField(max_length=160)
    og_description = models.TextField()
    share_count = models.IntegerField(default=0)
    
    def get_absolute_url(self):
        return f"/ref/{self.student.username}-{self.city}/"

# Auto-create SEO content on referral signup
@receiver(post_save, sender=ReferralPage)
def generate_ref_content(sender, instance, created, **kwargs):
    if created:
        generate_referral_seo(instance)
